"""
EROS 3.0 Decision Pipeline.

This is orchestration only.

The pipeline does not duplicate:
- data provider mathematics
- quality mathematics
- scoring mathematics
- classification rules
- SOTP mathematics
- DCF mathematics
- Stage 13C mathematics
- Stage 14 mathematics

Those remain owned by their respective modules.

EROS coordinates them and preserves provenance.
"""

from dataclasses import dataclass, field
from typing import Any, Optional


@dataclass
class PipelineContext:
    symbol: str

    financials: Any = None
    normalized: Any = None
    quality: Any = None
    score: Any = None
    classification: Any = None

    sotp: Any = None
    dcf: Any = None
    valuation_bridge: Any = None

    stage13c: Any = None
    stage14: Any = None

    evidence: list = field(default_factory=list)
    warnings: list = field(default_factory=list)

    status: str = "INITIALIZED"


def create_context(symbol: str) -> PipelineContext:
    if not symbol or not str(symbol).strip():
        raise ValueError("symbol is required")

    return PipelineContext(
        symbol=str(symbol).strip().upper()
    )


def attach(context: PipelineContext, name: str, value: Any):
    setattr(context, name, value)
    return context


def add_warning(context: PipelineContext, message: str):
    if message:
        context.warnings.append(str(message))


def add_evidence(context: PipelineContext, evidence: Any):
    if evidence is not None:
        context.evidence.append(evidence)


def finalize(context: PipelineContext):
    context.status = "COMPLETE"

    return {
        "symbol": context.symbol,
        "status": context.status,

        "financials": context.financials,
        "normalized": context.normalized,
        "quality": context.quality,
        "score": context.score,
        "classification": context.classification,

        "sotp": context.sotp,
        "dcf": context.dcf,
        "valuation_bridge": context.valuation_bridge,

        "stage13c": context.stage13c,
        "stage14": context.stage14,

        "evidence": context.evidence,
        "warnings": context.warnings,

        "provenance": {
            "orchestrator": "EROS_3.0",
            "symbol": context.symbol,
        },
    }


__all__ = [
    "PipelineContext",
    "create_context",
    "attach",
    "add_warning",
    "add_evidence",
    "finalize",
]


===== API SERVICE =====
"""
EROS 3.0 public service facade.

The facade currently performs orchestration validation.
Individual engines are integrated behind explicit boundaries.
"""

from eros.decision.pipeline import (
    create_context,
    finalize,
)


def evaluate(symbol: str, **kwargs):

    context = create_context(symbol)

    # Inputs are attached without changing their semantics.
    for key, value in kwargs.items():
        if hasattr(context, key):
            setattr(context, key, value)

    return finalize(context)


__all__ = ["evaluate"]
